cmake_minimum_required(VERSION 3.22)
project(framezoo_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(NATIVE_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/.work/prefix/${ANDROID_ABI}")
if(NOT EXISTS "${NATIVE_PREFIX}/include/libtorrent/libtorrent.hpp" OR NOT EXISTS "${NATIVE_PREFIX}/include/mpv/client.h")
    message(FATAL_ERROR "Thiếu libtorrent/libmpv. Kiểm tra tác vụ Gradle buildNative và native/build-native.sh.")
endif()
add_library(libtorrent STATIC IMPORTED)
set_target_properties(libtorrent PROPERTIES IMPORTED_LOCATION "${NATIVE_PREFIX}/lib/libtorrent-rasterbar.a")
add_library(ssl STATIC IMPORTED)
set_target_properties(ssl PROPERTIES IMPORTED_LOCATION "${NATIVE_PREFIX}/lib/libssl.a")
add_library(crypto STATIC IMPORTED)
set_target_properties(crypto PROPERTIES IMPORTED_LOCATION "${NATIVE_PREFIX}/lib/libcrypto.a")
add_library(mpv SHARED IMPORTED)
set_target_properties(mpv PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libmpv.so")
add_library(avcodec SHARED IMPORTED)
set_target_properties(avcodec PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libavcodec.so")

add_library(framezoo_native SHARED TorrentStream.cpp MpvBridge.cpp)
set(MPV_ARCH "arm64")
if(ANDROID_ABI STREQUAL "x86_64")
    set(MPV_ARCH "x86_64")
endif()
target_include_directories(framezoo_native PRIVATE "${NATIVE_PREFIX}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/.work/src/boost_1_86_0" "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/.work/src/mpv-android/buildscripts/prefix/${MPV_ARCH}/include")
target_compile_definitions(framezoo_native PRIVATE TORRENT_ABI_VERSION=100 TORRENT_USE_OPENSSL TORRENT_USE_LIBCRYPTO TORRENT_USE_RTC=0)
target_link_libraries(framezoo_native PRIVATE libtorrent ssl crypto mpv avcodec android log z)
